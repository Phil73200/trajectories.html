let seasonIndex = 1; rankCols.forEach(col => { const rank = Number(r[col]); if(Number.isFinite(rank) && rank > 0){ if(!buckets[seasonIndex]){ buckets[seasonIndex] = []; } buckets[seasonIndex].push(rank);
